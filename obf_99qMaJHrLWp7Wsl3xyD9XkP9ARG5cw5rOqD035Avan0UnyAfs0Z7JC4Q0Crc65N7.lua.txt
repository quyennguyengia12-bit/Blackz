-- ===================== 1. ANTI-KICK & ANTI-TELEPORT =====================
local function InitAntiCheat()
    local Players = game:GetService("Players")
    local LP = Players.LocalPlayer
    
    if hookmetamethod and newcclosure and getnamecallmethod then
        pcall(function()
            local oldNamecall
            oldNamecall = hookmetamethod(game, "__namecall", newcclosure(function(self, ...)
                local method = getnamecallmethod()
                if self == LP and (method == "Kick" or method == "kick" or method == "Teleport") then
                    warn("⚠️ [Anti-Cheat] Chặn " .. method .. "!")
                    return nil
                end
                return oldNamecall(self, ...)
            end))
        end)
    end
    
    pcall(function()
        local mt = getrawmetatable(game)
        if mt and setreadonly then
            setreadonly(mt, false)
            local oldIndex = mt.__index
            mt.__index = newcclosure(function(self, key)
                if self == LP and key == "Kick" then return function() end end
                return oldIndex(self, key)
            end)
            setreadonly(mt, true)
        end
    end)
    
    Players.PlayerRemoving:Connect(function(player)
        if player == LP then warn("⚠️ [Anti-Cheat] Đang cố giữ kết nối...") end
    end)
end
pcall(InitAntiCheat)

local function CreateSafeScreenGui(name)
    local gui = Instance.new("ScreenGui")
    gui.Name = name
    gui.ResetOnSpawn = false
    gui.IgnoreGuiInset = true
    gui.DisplayOrder = 999
    gui.Parent = PlayerGui
    return gui

local function safeClick(obj)
    if not obj then return end
    pcall(function()
        local cd = obj:FindFirstChildWhichIsA("ClickDetector")
        if cd then
            if fireclickdetector then fireclickdetector(cd) else cd:Click() end
            return
        end
        for _, child in ipairs(obj:GetDescendants()) do
            if child:IsA("ClickDetector") then
                if fireclickdetector then fireclickdetector(child) else child:Click() end
                return
            end
        end
    end)
end

local function getEggName(egg)
    if egg:GetAttribute("Type") then return tostring(egg:GetAttribute("Type")) end
    if egg:GetAttribute("Name") then return tostring(egg:GetAttribute("Name")) end
    return egg.Name
end

local function isValidEgg(egg)
    if not egg or not egg.Parent then return false end
    if egg:IsA("BasePart") and (egg.Transparency == 1 or not egg.CanCollide) then return false end
    if egg:GetAttribute("Collected") or egg:GetAttribute("Stolen") then return false end
    return true
end

local function getEggValue(egg)
    local name = getEggName(egg)
    return EGG_DATABASE[name] and EGG_DATABASE[name].value or 0
end

local function detectMutation(egg)
    local name = egg.Name:lower()
    if name:find("bloom") or name:find("spirit") or name:find("cosmic") then return true end
    if name:find("golden") or name:find("silver") then return true end
    if egg:GetAttribute("Mutation") or egg:GetAttribute("Bloom") then return true end
    return false
end

local function teleportTo(pos)
    if not HRP then return end
    HRP.CFrame = CFrame.new(pos + Vector3.new(0, CONFIG.TPHeightOffset, 0))
end

-- ===================== 6. CACHE OPTIMIZED =====================
local function updateCaches()
    if tick() - lastCacheUpdate < 3 then return end
    lastCacheUpdate = tick()
    
    eggCache = {}
    facilityCache = {}
    
    for _, obj in ipairs(Workspace:GetDescendants()) do
        if obj:IsA("BasePart") or obj:IsA("Model") then
            local nameLower = obj.Name:lower()
            if (nameLower:find("egg") or obj:GetAttribute("Egg")) and isValidEgg(obj) then
                table.insert(eggCache, obj)
            elseif nameLower:find("sell") then
                table.insert(facilityCache, {obj = obj, type = "sell"})
            elseif nameLower:find("hatch") or nameLower:find("nest") then
                table.insert(facilityCache, {obj = obj, type = "hatch"})
            elseif nameLower:find("upgrade") then
                table.insert(facilityCache, {obj = obj, type = "upgrade"})
            elseif nameLower:find("claim") or nameLower:find("reward") then
                table.insert(facilityCache, {obj = obj, type = "claim"})
            elseif nameLower:find("rebirth") or nameLower:find("prestige") then
                table.insert(facilityCache, {obj = obj, type = "rebirth"})
            elseif nameLower:find("treadmill") then
                table.insert(facilityCache, {obj = obj, type = "treadmill"})
            elseif nameLower:find("place") then
                table.insert(facilityCache, {obj = obj, type = "place"})
            elseif nameLower:find("fuse") then
                table.insert(facilityCache, {obj = obj, type = "fuse"})
            end
        end
    end
end

local function findFacility(facType)
    for _, f in ipairs(facilityCache) do
        if f.type == facType then return f.obj end
    end
    return nil
end

-- ===================== 7. FLY SYSTEM =====================
local function updateFly()
    if not State.Fly and not State.LowFly then
        if FlyBodyVelocity then FlyBodyVelocity:Destroy(); FlyBodyVelocity = nil end
        if Hum then Hum.PlatformStand = false; Hum.AutoRotate = true end
        return
    end
    if not HRP then return end
    if not FlyBodyVelocity then
        FlyBodyVelocity = Instance.new("BodyVelocity")
        FlyBodyVelocity.MaxForce = Vector3.new(1e6, 1e6, 1e6)
        FlyBodyVelocity.P = 1e5
        FlyBodyVelocity.Parent = HRP
    end
    if Hum then Hum.PlatformStand = true; Hum.AutoRotate = false end
    
    local direction = Vector3.zero
    if State.Fly then
        local moveDir = Hum.MoveDirection
        local vertical = (FlyUpInput and 1 or 0) - (FlyDownInput and 1 or 0)
        direction = Vector3.new(moveDir.X, vertical, moveDir.Z)
    end
    FlyBodyVelocity.Velocity = direction.Magnitude > 0 and direction.Unit * CONFIG.FlySpeed or Vector3.zero
end

-- ===================== 8. MAIN LOOP =====================
local lastAction = { steal=0, hatch=0, sell=0, upgrade=0, claim=0, rebirth=0, afk=0, treadmill=0, place=0, fuse=0 }

RunService.Heartbeat:Connect(function(dt)
    if not HRP or not Hum then return end
    local now = tick()
    
    updateCaches()
    updateFly()
    
    if State.SpeedBypass then Hum.WalkSpeed = 100 else Hum.WalkSpeed = CONFIG.WalkSpeed end
    if State.InfiniteJump then Hum.JumpPower = 100 end
    if State.Noclip then for _, p in ipairs(Char:GetDescendants()) do if p:IsA("BasePart") then p.CanCollide = false end end end
    if State.WalkWater and HRP.Position.Y < 0 then HRP.CFrame = CFrame.new(HRP.Position.X, 5, HRP.Position.Z) end
    if State.Godmode then pcall(function() if Hum.Health < Hum.MaxHealth then Hum.Health = Hum.MaxHealth end end) end
    if State.Fullbright then pcall(function() Lighting.Brightness = 2; Lighting.GlobalShadows = false end) else pcall(function() Lighting.Brightness = 1; Lighting.GlobalShadows = true end) end
    
    -- Auto Steal (Smart)
    if State.AutoSteal and now - lastAction.steal > CONFIG.StealDelay then
        lastAction.steal = now
        task.spawn(function()
            local bestEgg, bestValue = nil, -1
            for _, egg in ipairs(eggCache) do
                local val = getEggValue(egg)
                if State.RareHunter then
                    local data = EGG_DATABASE[getEggName(egg)]
                    if data and data.rarity == "Legendary" or data and data.rarity == "Mythic" or data and data.rarity == "Secret" or data and data.rarity == "Cosmic" or data and data.rarity == "Eternal" or data and data.rarity == "Divine" then
                        val = val * 10
                    end
                end
                if State.MutationPriority and detectMutation(egg) then val = val * 5 end
                if State.SizeHunter then val = val * (1 + egg.Size.Magnitude / 10) end
                if val > bestValue and val >= CONFIG.MinValueToSteal then
                    bestValue = val; bestEgg = egg
                end
            end
            if bestEgg then
                if State.AutoTP then
                    local pos = bestEgg:IsA("Model") and bestEgg:GetPivot().Position or bestEgg.Position
                    teleportTo(pos)
                end
                task.wait(0.2)
                safeClick(bestEgg)
                task.wait(0.1)
                safeClick(bestEgg)
            end
        end)
    end
    
    -- Auto Hatch
    if State.AutoHatch and now - lastAction.hatch > CONFIG.HatchDelay then
        lastAction.hatch = now
        local hatch = findFacility("hatch")
        if hatch then safeClick(hatch) end
    end
    
    -- Auto Sell
    if State.AutoSell and now - lastAction.sell > CONFIG.SellInterval then
        lastAction.sell = now
        local sell = findFacility("sell")
        if sell then safeClick(sell) end
    end
    
    -- Auto Upgrade
    if State.AutoUpgrade and now - lastAction.upgrade > CONFIG.UpgradeInterval then
        lastAction.upgrade = now
        local upgrade = findFacility("upgrade")
        if upgrade then safeClick(upgrade) end
    end
    
    -- Auto Claim
    if State.AutoClaim and now - lastAction.claim > CONFIG.ClaimInterval then
        lastAction.claim = now
        local claim = findFacility("claim")
        if claim then safeClick(claim) end
    end
    
    -- Auto Rebirth
    if State.AutoRebirth and now - lastAction.rebirth > CONFIG.RebirthInterval then
        lastAction.rebirth = now
        local rebirth = findFacility("rebirth")
        if rebirth then safeClick(rebirth) end
    end
    
    -- Auto Treadmill
    if State.AutoTreadmill and now - lastAction.treadmill > 20 then
        lastAction.treadmill = now
        local treadmill = findFacility("treadmill")
        if treadmill then safeClick(treadmill) end
    end
    
    -- Auto Place
    if State.AutoPlace and now - lastAction.place > 10 then
        lastAction.place = now
        local place = findFacility("place")
        if place then safeClick(place) end
    end
    
    -- Auto Fuse
    if State.AutoFuse and now - lastAction.fuse > 60 then
        lastAction.fuse = now
        local fuse = findFacility("fuse")
        if fuse then safeClick(fuse) end
    end
    
    -- Anti AFK
    if State.AntiAFK and now - lastAction.afk > CONFIG.AFKInterval then
        lastAction.afk = now
        HRP.CFrame = CFrame.new(HRP.Position + Vector3.new(math.random(-2,2), 0, math.random(-2,2)))
    end
end)

        Instance.new("UICorner", btn).CornerRadius = UDim.new(0, 10)
        Instance.new("UIStroke", btn).Color = Color3.fromRGB(0, 255, 150)
        return btn
    end
    
    local UpBtn = makeBtn("⬆", 0.6)
    local DownBtn = makeBtn("⬇", 0.75)
    
    UpBtn.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then
            FlyUpInput = true
        end
    end)
    UpBtn.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then
            FlyUpInput = false
        end
    end)
    DownBtn.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then
            FlyDownInput = true
        end
    end)
    DownBtn.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then
            FlyDownInput = false
        end
    end)
    
    RunService.Heartbeat:Connect(function()
        if FlyControlsGui then FlyControlsGui.Enabled = State.Fly end
    end)
end

-- ===================== 10. ESP SYSTEM =====================
local ESPContainer = Instance.new("Folder")
ESPContainer.Name = "SE_ESP"
ESPContainer.Parent = PlayerGui

local function clearESP()
    for _, tag in ipairs(ESPContainer:GetChildren()) do tag:Destroy() end
end

local function updateESP()
    if not State.ESP then clearESP(); return end
    clearESP()
    if not HRP then return end
    
    local nearby = {}
    for _, egg in ipairs(eggCache) do
        if isValidEgg(egg) then
            local pos = egg:IsA("Model") and egg:GetPivot().Position or egg.Position
            local dist = (HRP.Position - pos).Magnitude
            if dist < 400 then
                local name = getEggName(egg)
                local data = EGG_DATABASE[name]
                local val = data and data.value or 0
                local rarity = data and data.rarity or "Unknown"
                table.insert(nearby, {egg = egg, val = val, rarity = rarity, name = name})
            end
        end
