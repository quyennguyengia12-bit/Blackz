local function getEggName(egg)
    if egg:GetAttribute("Type") then return tostring(egg:GetAttribute("Type")) end
    if egg:GetAttribute("Name") then return tostring(egg:GetAttribute("Name")) end
    if egg.Parent and egg.Parent ~= Workspace and egg.Parent.Name ~= "Eggs" then return egg.Parent.Name end
    return egg.Name
end

local function isValidEgg(egg)
    if not egg or not egg.Parent then return false end
    if egg:IsA("BasePart") and (egg.Transparency == 1 or not egg.CanCollide) then return false end
    if egg:GetAttribute("Collected") or egg:GetAttribute("Stolen") or egg:GetAttribute("Claimed") then return false end
    return true
end

local function getPlayerSpeed()
    if Hum then return Hum.WalkSpeed end
    return 0
end

local function detectMutation(egg)
    local name = egg.Name:lower()
    if name:find("bloom") or name:find("spirit") or name:find("cosmic") or name:find("golden") then return true end
    if egg:GetAttribute("Mutation") or egg:GetAttribute("Bloom") then return true end
    return false
end

local clickCount = 0
local clickResetTime = 0

local function canClick()
    local now = tick()
    if now - clickResetTime > 1 then
        clickCount = 0
        clickResetTime = now
    end
    if clickCount < CONFIG.MaxClickPerSecond then
        clickCount = clickCount + 1
        return true
    end
    return false
end

local function safeClick(obj)
    if not obj or not canClick() then return end
    local now = tick()
    if now - lastSafeClick < clickCooldown then return end
    lastSafeClick = now
    pcall(function()
        local cd = obj:FindFirstChildWhichIsA("ClickDetector")
        if cd then
            if fireclickdetector then fireclickdetector(cd) else cd:Click() end
            return
        end
        for _, child in ipairs(obj:GetDescendants()) do
            if child:IsA("ClickDetector") then
                if fireclickdetector then fireclickdetector(child) else child:Click() end
                return
            end
        end
        local prompt = obj:FindFirstChildWhichIsA("ProximityPrompt")
        if prompt then
            if fireproximityprompt then fireproximityprompt(prompt) else prompt:InputHoldBegin() task.wait(0.1) prompt:InputHoldEnd() end
            return
        end
    end)
end

local function teleportTo(pos)
    if not HRP then return end
    local now = tick()
    if now - lastTeleportTime < 1 then return end
    lastTeleportTime = now
    if (HRP.Position - pos).Magnitude < 5 then return end
    HRP.CFrame = CFrame.new(pos + Vector3.new(0, CONFIG.TPHeightOffset, 0))
end

local function getBestBiome()
    local speed = getPlayerSpeed()
    local best = "Forest"
    for biome, data in pairs(BIOME_DATA) do
        if speed >= data.speedRequired then
            best = biome
        end
    end
    return best
end

-- ===================== CACHE =====================
local function updateCaches()
    if tick() - lastUpdate < 5 then return end
    lastUpdate = tick()
    eggCache = {}
    facilityCache = {}
    guardianCache = {}
    for _, obj in ipairs(Workspace:GetDescendants()) do
        if obj:IsA("BasePart") or obj:IsA("Model") then
            local nameLower = obj.Name:lower()
            if (nameLower:find("egg") or obj:GetAttribute("Egg")) and isValidEgg(obj) then
                table.insert(eggCache, obj)
            elseif nameLower:find("guardian") or obj:GetAttribute("Guardian") then
                table.insert(guardianCache, obj)
            elseif nameLower:find("sell") then
                table.insert(facilityCache, { obj = obj, type = "sell" })
            elseif nameLower:find("hatch") or nameLower:find("nest") or nameLower:find("incubator") then
                table.insert(facilityCache, { obj = obj, type = "hatch" })
            elseif nameLower:find("upgrade") then
                table.insert(facilityCache, { obj = obj, type = "upgrade" })
            elseif nameLower:find("claim") or nameLower:find("reward") or nameLower:find("collect") then
                table.insert(facilityCache, { obj = obj, type = "claim" })
            elseif nameLower:find("rebirth") or nameLower:find("prestige") then
                table.insert(facilityCache, { obj = obj, type = "rebirth" })
            elseif nameLower:find("treadmill") then
                table.insert(facilityCache, { obj = obj, type = "treadmill" })
            elseif nameLower:find("place") then
                table.insert(facilityCache, { obj = obj, type = "place" })
            elseif nameLower:find("fuse") or nameLower:find("combine") or nameLower:find("merge") then
                table.insert(facilityCache, { obj = obj, type = "fuse" })
            end
        end
    end
end

local function findFacility(facType)
    for _, f in ipairs(facilityCache) do
        if f.type == facType then return f.obj end
    end
    return nil
end

-- ===================== FLY SYSTEM =====================
local function updateFly()
    if not State.Fly then
        if FlyBodyVelocity then FlyBodyVelocity:Destroy(); FlyBodyVelocity = nil end
        if Hum then Hum.PlatformStand = false; Hum.AutoRotate = true end
        return
    end
    if not HRP then return end
    if not FlyBodyVelocity then
        FlyBodyVelocity = Instance.new("BodyVelocity")
        FlyBodyVelocity.MaxForce = Vector3.new(1e6, 1e6, 1e6)
        FlyBodyVelocity.P = 1e5
        FlyBodyVelocity.Parent = HRP
    end
    if Hum then Hum.PlatformStand = true; Hum.AutoRotate = false end
    local moveDir = Hum.MoveDirection
    local vertical = (FlyUpInput and 1 or 0) - (FlyDownInput and 1 or 0)
    local direction = Vector3.new(moveDir.X, vertical, moveDir.Z)
    FlyBodyVelocity.Velocity = direction.Magnitude > 0 and direction.Unit * CONFIG.FlySpeed or Vector3.zero
end

        Instance.new("UIStroke", btn).Color = Color3.fromRGB(0, 255, 150)
        return btn
    end
    local UpBtn = makeBtn("⬆", 0.6)
    local DownBtn = makeBtn("⬇", 0.75)
    UpBtn.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then FlyUpInput = true end
    end)
    UpBtn.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then FlyUpInput = false end
    end)
    DownBtn.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then FlyDownInput = true end
    end)
    DownBtn.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then FlyDownInput = false end
    end)
    RunService.Heartbeat:Connect(function()
        if FlyControlsGui then FlyControlsGui.Enabled = State.Fly end
    end)
end

-- ===================== MAIN LOOP =====================
local lastAction = { steal = 0, hatch = 0, sell = 0, upgrade = 0, claim = 0, rebirth = 0, afk = 0, treadmill = 0, place = 0, fuse = 0, serverHop = 0 }

local function StartMainLoop()
    RunService.Heartbeat:Connect(function(dt)
        if not HRP or not Hum then return end
        local now = tick()
        currentSpeed = getPlayerSpeed()
        updateCaches()
        updateFly()

        if CONFIG.SelectedBiome == "Auto" then
            currentBiome = getBestBiome()
        else
            currentBiome = CONFIG.SelectedBiome
        end

        local biomeData = BIOME_DATA[currentBiome]
        if biomeData and currentSpeed < biomeData.speedRequired then
            if CONFIG.SelectedBiome == "Auto" then
                currentBiome = getBestBiome()
            else
                currentBiome = nil
            end
        end

        if State.SpeedBypass then Hum.WalkSpeed = 100 else Hum.WalkSpeed = CONFIG.WalkSpeed end
        if State.InfiniteJump then Hum.JumpPower = 100 end
        if Char and State.Noclip then
            for _, part in ipairs(Char:GetDescendants()) do
                if part:IsA("BasePart") then part.CanCollide = false end
            end
        end
        if State.WalkWater and HRP and HRP.Position.Y < 0 then HRP.CFrame = CFrame.new(HRP.Position.X, 5, HRP.Position.Z) end
        if State.Godmode then pcall(function() if Hum.Health < Hum.MaxHealth then Hum.Health = Hum.MaxHealth end end) end
        if State.Fullbright then pcall(function() Lighting.Brightness = 2; Lighting.GlobalShadows = false end) else pcall(function() Lighting.Brightness = 1; Lighting.GlobalShadows = true end) end

        -- Auto Steal
        if State.AutoSteal and currentBiome and now - lastAction.steal > CONFIG.StealDelay then
            lastAction.steal = now
            task.spawn(function()
                local bestEgg, bestScore = nil, -1
                for _, egg in ipairs(eggCache) do
                    if isValidEgg(egg) then
                        local name = getEggName(egg)
                        local data = EGG_DATABASE[name]
                        if data and (data.biome == currentBiome or data.isMutation) then
                            if currentSpeed < data.speedRequired then continue end
                            local val = data.value or 0
                            local score = val
                            if State.RareHunter then
                                local rarityPriority = { Common = 1, Uncommon = 2, Rare = 5, Epic = 10, Legendary = 25, Mythic = 50, Secret = 100, Cosmic = 250, Eternal = 500, Divine = 1000 }
                                score = score * (rarityPriority[data.rarity] or 1)
                            end
                            if State.MutationPriority and detectMutation(egg) then score = score * 5 end
                            if score > bestScore and val >= CONFIG.MinValueToSteal then
                                bestScore = score
                                bestEgg = egg
                            end
                        end
                    end
                end
                if bestEgg then
                    if State.EnableGuardianAvoid then
                        local nearestGuardian, minDist = nil, 30
                        for _, g in ipairs(guardianCache) do
                            if g and g:IsA("Model") then
                                local pos = g:GetPivot().Position
                                local dist = (HRP.Position - pos).Magnitude
                                if dist < minDist then minDist = dist; nearestGuardian = g end
                            end
                        end
                        if nearestGuardian then
                            if State.GuardianFreeze then
                                pcall(function()
                                    for _, part in ipairs(nearestGuardian:GetDescendants()) do
                                        if part:IsA("BasePart") then part.CanCollide = false; part.Anchored = true end
                                    end
                                    nearestGuardian:SetPrimaryPartCFrame(nearestGuardian:GetPivot() * CFrame.new(0, -100, 0))
                                end)
                            else
                                local dir = (HRP.Position - nearestGuardian:GetPivot().Position).Unit
                                teleportTo(HRP.Position + dir * 50)
                                task.wait(1)
                                return
                            end
                        end
                    end
                    if State.AutoTP then
                        local pos = bestEgg:IsA("Model") and bestEgg:GetPivot().Position or bestEgg.Position
                        teleportTo(pos)
                    end
                    task.wait(0.2)
                    safeClick(bestEgg)
                end
            end)
        end

        -- Auto Hatch
        if State.AutoHatch and now - lastAction.hatch > CONFIG.HatchDelay then
            lastAction.hatch = now
            task.spawn(function() local h = findFacility("hatch"); if h then safeClick(h) end end)
        end
        if State.AutoSell and now - lastAction.sell > CONFIG.SellInterval then
            lastAction.sell = now
            task.spawn(function() local s = findFacility("sell"); if s then safeClick(s) end end)
        end
        if State.AutoUpgrade and now - lastAction.upgrade > CONFIG.UpgradeInterval then
            lastAction.upgrade = now
            task.spawn(function() local u = findFacility("upgrade"); if u then safeClick(u) end end)
        end
        if State.AutoClaim and now - lastAction.claim > CONFIG.ClaimInterval then
            lastAction.claim = now
            task.spawn(function() local c = findFacility("claim"); if c then safeClick(c) end end)
        end
        if State.AutoRebirth and now - lastAction.rebirth > CONFIG.RebirthInterval then
            lastAction.rebirth = now
            task.spawn(function() local r = findFacility("rebirth"); if r then safeClick(r) end end)
        end
        if State.AutoTreadmill and now - lastAction.treadmill > 30 then
            lastAction.treadmill = now
            task.spawn(function() local t = findFacility("treadmill"); if t then safeClick(t) end end)
        end
        if State.AutoPlace and now - lastAction.place > 15 then
            lastAction.place = now
            task.spawn(function() local p = findFacility("place"); if p then safeClick(p) end end)
        end
        if State.AutoFuse and now - lastAction.fuse > 60 then
            lastAction.fuse = now
            task.spawn(function() local f = findFacility("fuse"); if f then safeClick(f) end end)
        end

        if State.AutoEquipBest then
            pcall(function()
                for _, gui in ipairs(PlayerGui:GetDescendants()) do
                    if gui:IsA("TextButton") and gui.Name:lower():find("equip") and gui.Visible then
                        local cd = gui:FindFirstChildWhichIsA("ClickDetector")
                        if cd then cd:Click() end
                        break
                    end
                end
            end)
        end

        if State.AntiAFK and now - lastAction.afk > CONFIG.AFKInterval then
            lastAction.afk = now
            if HRP then HRP.CFrame = CFrame.new(HRP.Position + Vector3.new(math.random(-2,2), 0, math.random(-2,2))) end
        end

        if CONFIG.EnableServerHop and State.AutoSteal then
            local eggCount = 0
            for _, egg in ipairs(eggCache) do
                if isValidEgg(egg) then
                    local data = EGG_DATABASE[getEggName(egg)]
                    if data and data.biome == currentBiome then eggCount = eggCount + 1 end
                end
            end
            if eggCount < 3 and now - lastAction.serverHop > 120 then
                lastAction.serverHop = now
                warn("⚠️ Ít trứng ở " .. currentBiome .. ", đang chuyển server...")
                pcall(function()
                    local servers = {}
                    local req = game:HttpGet("https://games.roblox.com/v1/games/" .. game.PlaceId .. "/servers/Public?sortOrder=Asc&limit=100")
                    local data = HttpService:JSONDecode(req)
                    if data and data.data then
                        for _, s in ipairs(data.data) do
                            if s.playing < s.maxPlayers and s.id ~= game.JobId then
                                table.insert(servers, s.id)
                            end
                        end
                    end
                    if #servers > 0 then
                        TeleportService:TeleportToPlaceInstance(game.PlaceId, servers[math.random(1, #servers)], LP)
                    end
                end)
            end
        end
    end)
end

local function UpdateESP()
    pcall(function()
        if not ESPContainer then
            ESPContainer = Instance.new("Folder")
            ESPContainer.Name = "SE_ESP_Container"
            ESPContainer.Parent = CoreGui
        end
        for _, child in ipairs(ESPContainer:GetChildren()) do child:Destroy() end
        if not State.ESP then return end

        for _, egg in ipairs(eggCache) do
            if isValidEgg(egg) then
                local billboard = Instance.new("BillboardGui")
                billboard.Size = UDim2.new(0, 40, 0, 20)
                billboard.AlwaysOnTop = true
                billboard.Adornee = egg:IsA("Model") and egg.PrimaryPart or egg
                billboard.Parent = ESPContainer
                local lbl = Instance.new("TextLabel", billboard)
                lbl.Size = UDim2.new(1, 0, 1, 0)
                lbl.BackgroundTransparency = 1
                lbl.Text = "🥚"
                lbl.TextColor3 = Color3.fromRGB(0, 255, 150)
                lbl.TextSize = 10
                lbl.Font = Enum.Font.GothamBold
            end
        end
        for _, g in ipairs(guardianCache) do
            if g and g:IsA("Model") then
                local billboard = Instance.new("BillboardGui")
                billboard.Size = UDim2.new(0, 50, 0, 20)
                billboard.AlwaysOnTop = true
                billboard.Adornee = g.PrimaryPart or g:FindFirstChildWhichIsA("BasePart")

                local lbl = Instance.new("TextLabel", billboard)
                lbl.Size = UDim2.new(1, 0, 1, 0)
                lbl.BackgroundTransparency = 1
                lbl.Text = "⚠️"
                lbl.TextColor3 = Color3.fromRGB(255, 50, 50)
                lbl.TextSize = 10
                lbl.Font = Enum.Font.GothamBold
